// ============================================================ // 👥 USERS MANAGEMENT ENDPOINTS // ============================================================ // الحصول على جميع المستخدمين app.get('/api/users', authenticate, (req, res) => { db.all('SELECT id, username, name, role, two_fa_enabled, status, created_at FROM users', (err, users) => { if (err) { return res.status(500).json({ success: false, error: err.message }); } res.json({ success: true, users: users || [] }); }); }); // إضافة مستخدم جديد app.post('/api/users', authenticate, async (req, res) => { const { name, username, password, role, two_fa_enabled } = req.body; if (!name || !username || !password) { return res.status(400).json({ success: false, error: 'جميع الحقول مطلوبة' }); } try { // التحقق من وجود المستخدم const existing = await new Promise((resolve, reject) => { db.get('SELECT id FROM users WHERE username = ?', [username], (err, row) => { if (err) reject(err); else resolve(row); }); }); if (existing) { return res.status(400).json({ success: false, error: 'اسم المستخدم موجود بالفعل' }); } const hashedPassword = bcrypt.hashSync(password, 10); const twoFaSecret = two_fa_enabled ? speakeasy.generateSecret().base32 : null; const result = await new Promise((resolve, reject) => { db.run( 'INSERT INTO users (name, username, password, role, two_fa_secret, two_fa_enabled, status) VALUES (?, ?, ?, ?, ?, ?, ?)', [name, username, hashedPassword, role || 'viewer', twoFaSecret, two_fa_enabled ? 1 : 0, 'active'], function(err) { if (err) reject(err); else resolve(this); } ); }); logActivity(req.userId, 'create_user', `Created user: ${username}`); res.json({ success: true, id: result.lastID }); } catch (error) { console.error('❌ Error creating user:', error); res.status(500).json({ success: false, error: 'حدث خطأ في الخادم' }); } }); // تحديث مستخدم app.put('/api/users/:id', authenticate, async (req, res) => { const { id } = req.params; const { name, username, password, role, two_fa_enabled, status } = req.body; try { let query = 'UPDATE users SET name = ?, username = ?, role = ?, two_fa_enabled = ?, status = ?'; const params = [name, username, role, two_fa_enabled ? 1 : 0, status || 'active']; if (password) { const hashedPassword = bcrypt.hashSync(password, 10); query += ', password = ?'; params.push(hashedPassword); } if (two_fa_enabled) { const secret = speakeasy.generateSecret().base32; query += ', two_fa_secret = ?'; params.push(secret); } query += ' WHERE id = ?'; params.push(id); await new Promise((resolve, reject) => { db.run(query, params, function(err) { if (err) reject(err); else resolve(this); }); }); logActivity(req.userId, 'update_user', `Updated user: ${username}`); res.json({ success: true }); } catch (error) { console.error('❌ Error updating user:', error); res.status(500).json({ success: false, error: 'حدث خطأ في الخادم' }); } }); // حذف مستخدم app.delete('/api/users/:id', authenticate, (req, res) => { const { id } = req.params; // منع حذف المستخدم الحالي if (id == req.userId) { return res.status(400).json({ success: false, error: 'لا يمكن حذف حسابك الحالي' }); } db.run('DELETE FROM users WHERE id = ?', [id], function(err) { if (err) { return res.status(500).json({ success: false, error: err.message }); } if (this.changes === 0) { return res.status(404).json({ success: false, error: 'المستخدم غير موجود' }); } logActivity(req.userId, 'delete_user', `Deleted user ID: ${id}`); res.json({ success: true }); }); });
