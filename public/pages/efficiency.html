${good}

${bad}

${maint}

📊 الجاهزية

`; } // ============================================================ // 1. الجدول العام // ============================================================ function renderGeneralEfficiencyTable(vessels) { const container = document.getElementById('generalEffTableContainer'); if (!container) { console.error('❌ generalEffTableContainer not found'); return; } if (!vessels || vessels.length === 0) { container.innerHTML = `

`; sorted.forEach((v, index) => { const statusColor = v.stat === 'صالح' ? '#28a745' : v.stat === 'معطب' ? '#dc3545' : '#ffc107'; const statusIcon = v.stat === 'صالح' ? '✅' : v.stat === 'معطب' ? `; container.innerHTML = html; } // ============================================================ // 2. جدول الفئات (نفس خصائص الجدول العام) // ============================================================ function renderCategoryEfficiencyTable(vessels) { const container = document.getElementById('categoryEffContainer'); if (!container) return; const categories = ['البروق', 'صقور', 'خوافر', 'طوافات', 'زوارق مزدوجة']; let allHtml = ''; let totalAll = 0, goodAll = 0, badAll = 0, maintAll = 0; categories.forEach(cat => { const catVessels = vessels.filter(v => v.cat === cat); const total = catVessels.length; const good = catVessels.filter(v => v.stat === 'صالح').length; const bad = catVessels.filter(v => v.stat === 'معطب').length; const maint = catVessels.filter(v => v.stat === 'صيانة').length; const eff = total > 0 ? Math.round((good / total) * 100) : 0; const color = eff >= 80 ? '#28a745' : eff >= 50 ? '#ffc107' : '#dc3545'; totalAll += total; goodAll += good; badAll += bad; maintAll += maint; allHtml += `

${i + 1} ${v.name || '-'} ${v.num || '-'} ${v.cat || '-'} ${v.reg || '-'} ${v.zone || '-'} ${statusIcon} ${v.stat} ${lastMaintDate}
${total > 0 ? `
📊 نسبة الجاهزية: ${eff}%
` : ''}
`; }); // إجمالي الفئات const totalEff = totalAll > 0 ? Math.round((goodAll / totalAll) * 100) : 0; const totalColor = totalEff >= 80 ? '#28a745' : totalEff >= 50 ? '#ffc107' : '#dc3545'; allHtml += `

${regionVessels.length === 0 ? ` ` : regionVessels.map((v, i) => { const statusColor = v.stat === 'صالح' ? '#28a745' : v.stat === 'معطب' ? '#dc3545' : '#ffc107'; const statusIcon = v.stat === 'صالح' ? '✅' : v.stat === 'معطب' ? 🚫 لا توجد مراكب في هذا الإقليم

` : ''}
`; }); container.innerHTML = html; } // ============================================================ // دوال الفلترة // ============================================================ function showAllEfficiency() { renderEfficiency(); showAlert('✅ تم عرض جميع الجداول', 'success'); } function filterEfficiencyByUnit() { const select = document.getElementById('effUnitFilter'); if (!select) return; const value = select.value; if (value) { const filtered = allVessels.filter(v => v.reg === value); showAlert(`🔍 عرض إقليم: ${value}`, 'info'); updateEfficiencyStats(filtered); renderGeneralEfficiencyTable(filtered); renderCategoryEfficiencyTable(filtered); renderRegionEfficiencyTables(filtered); } else { showAlert('✅ عرض جميع الأقاليم', 'success'); renderEfficiency(); } } // ============================================================ // دالة viewVesselMaintenance // ============================================================ function viewVesselMaintenance(vesselId) { const vessel = allVessels.find(v => v.id == vesselId); if (!vessel) { showAlert('❌ المركب غير موجود', 'danger'); return; } const records = allMaintenance.filter(r => r.vesselId == vesselId); if (records.length === 0) { showAlert(`🚫 لا توجد سجلات صيانة للمركب: ${vessel.name}`, 'info'); return; } const modal = document.createElement('div'); modal.style.cssText = ` position: fixed; top: 0; left: 0; right: 0; bottom: 0; background: rgba(0,0,0,0.7); display: flex; justify-content: center; align-items: center; z-index: 99999; `; let html = `
