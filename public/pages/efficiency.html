// public/js/pages/efficiency.js console.log('✅ Efficiency page loaded'); // ============================================================ // دوال صفحة الجاهزية // ============================================================ function initEfficiency() { console.log('📊 Initializing efficiency page'); loadVessels(); } // ============================================================ // الدالة الرئيسية // ============================================================ function renderEfficiency() { console.log('📊 Rendering efficiency, vessels:', allVessels.length); const vessels = allVessels || []; // تحديث العدد const countEl = document.getElementById('effCount'); if (countEl) countEl.textContent = `📊 ${vessels.length} مركب`; // عرض الإحصائيات updateEfficiencyStats(vessels); // عرض الجدول العام renderGeneralEfficiencyTable(vessels); // عرض جدول الفئات renderCategoryEfficiencyTable(vessels); // عرض جداول الأقاليم renderRegionEfficiencyTables(vessels); } // ============================================================ // الإحصائيات // ============================================================ function updateEfficiencyStats(vessels) { const statsContainer = document.getElementById('statsCards'); if (!statsContainer) return; const total = vessels.length; const good = vessels.filter(v => v.stat === 'صالح').length; const bad = vessels.filter(v => v.stat === 'معطب').length; const maint = vessels.filter(v => v.stat === 'صيانة').length; const eff = total > 0 ? Math.round((good / total) * 100) : 0; statsContainer.innerHTML = `

${good}

✅ صالح

${bad}

❌ معطب

${maint}

🔧 صيانة

${eff}%

📊 الجاهزية

`; } // ============================================================ // 1. الجدول العام // ============================================================ function renderGeneralEfficiencyTable(vessels) { const container = document.getElementById('generalEffTableContainer'); if (!container) { console.error('❌ generalEffTableContainer not found'); return; } if (!vessels || vessels.length === 0) { container.innerHTML = `
🚫 لا توجد مراكب مسجلة
`; return; } const sorted = [...vessels].sort((a, b) => { const order = { 'صالح': 0, 'صيانة': 1, 'معطب': 2 }; return (order[a.stat] || 3) - (order[b.stat] || 3); }); let html = `
الجدول العام للمراكب (${vessels.length} مركب)
`; sorted.forEach((v, index) => { const statusColor = v.stat === 'صالح' ? '#28a745' : v.stat === 'معطب' ? '#dc3545' : '#ffc107'; const statusIcon = v.stat === 'صالح' ? '✅' : v.stat === 'معطب' ? '❌' : '🔧'; const lastMaintenance = allMaintenance .filter(r => r.vesselId === v.id) .sort((a, b) => new Date(b.date) - new Date(a.date))[0]; const lastMaintDate = lastMaintenance ? new Date(lastMaintenance.date).toLocaleDateString() : '-'; html += ` `; }); html += `
# المركب الرقم الفئة الإقليم المنطقة الحالة آخر صيانة الإجراءات
${index + 1} ${v.name || '-'} ${v.num || '-'} ${v.cat || '-'} ${v.reg || '-'} ${v.zone || '-'} ${statusIcon} ${v.stat} ${lastMaintDate}
`; container.innerHTML = html; } // ============================================================ // 2. جدول الفئات // ============================================================ function renderCategoryEfficiencyTable(vessels) { const container = document.getElementById('categoryEffContainer'); if (!container) return; const categories = ['البروق', 'صقور', 'خوافر', 'طوافات', 'زوارق مزدوجة']; let html = ''; categories.forEach(cat => { const catVessels = vessels.filter(v => v.cat === cat); const total = catVessels.length; const good = catVessels.filter(v => v.stat === 'صالح').length; const bad = catVessels.filter(v => v.stat === 'معطب').length; const maint = catVessels.filter(v => v.stat === 'صيانة').length; const eff = total > 0 ? Math.round((good / total) * 100) : 0; const color = eff >= 80 ? '#28a745' : eff >= 50 ? '#ffc107' : '#dc3545'; html += `
الفئة: ${cat} (${total} مركب) | ✅ ${good} | 🔧 ${maint} | ❌ ${bad} ${eff}% جاهزية
${catVessels.length === 0 ? ` ` : catVessels.map((v, i) => { const statusColor = v.stat === 'صالح' ? '#28a745' : v.stat === 'معطب' ? '#dc3545' : '#ffc107'; const statusIcon = v.stat === 'صالح' ? '✅' : v.stat === 'معطب' ? '❌' : '🔧'; const lastMaintenance = allMaintenance .filter(r => r.vesselId === v.id) .sort((a, b) => new Date(b.date) - new Date(a.date))[0]; const lastMaintDate = lastMaintenance ? new Date(lastMaintenance.date).toLocaleDateString() : '-'; return ` `; }).join('')}
# المركب الرقم الفئة الإقليم المنطقة الحالة آخر صيانة الإجراءات
🚫 لا توجد مراكب في هذه الفئة
${i + 1} ${v.name || '-'} ${v.num || '-'} ${v.cat || '-'} ${v.reg || '-'} ${v.zone || '-'} ${statusIcon} ${v.stat} ${lastMaintDate}
${total > 0 ? `
📊 نسبة الجاهزية: ${eff}%
` : ''}
`; }); container.innerHTML = html; } // ============================================================ // 3. جداول الأقاليم // ============================================================ function renderRegionEfficiencyTables(vessels) { const container = document.getElementById('regionsEffContainer'); if (!container) return; const regions = [ { name: '🗺️ الحرس البحري بالشمال', key: 'الشمال' }, { name: '🗺️ الحرس البحري بالساحل', key: 'الساحل' }, { name: '🗺️ الحرس البحري بالوسط', key: 'الوسط' }, { name: '🗺️ الحرس البحري بالجنوب', key: 'الجنوب' } ]; let html = ''; regions.forEach(region => { const regionVessels = vessels.filter(v => v.reg === region.key); const total = regionVessels.length; const good = regionVessels.filter(v => v.stat === 'صالح').length; const bad = regionVessels.filter(v => v.stat === 'معطب').length; const maint = regionVessels.filter(v => v.stat === 'صيانة').length; const eff = total > 0 ? Math.round((good / total) * 100) : 0; const color = eff >= 80 ? '#28a745' : eff >= 50 ? '#ffc107' : '#dc3545'; html += `
${region.name} 📊 ${total} مركب ✅ ${good} | 🔧 ${maint} | ❌ ${bad} ${eff}% جاهزية
${regionVessels.length === 0 ? ` ` : regionVessels.map((v, i) => { const statusColor = v.stat === 'صالح' ? '#28a745' : v.stat === 'معطب' ? '#dc3545' : '#ffc107'; const statusIcon = v.stat === 'صالح' ? '✅' : v.stat === 'معطب' ? '❌' : '🔧'; const lastMaintenance = allMaintenance .filter(r => r.vesselId === v.id) .sort((a, b) => new Date(b.date) - new Date(a.date))[0]; const lastMaintDate = lastMaintenance ? new Date(lastMaintenance.date).toLocaleDateString() : '-'; return ` `; }).join('')}
# المركب الرقم الفئة المنطقة الحالة آخر صيانة الإجراءات
🚫 لا توجد مراكب في هذا الإقليم
${i + 1} ${v.name || '-'} ${v.num || '-'} ${v.cat || '-'} ${v.zone || '-'} ${statusIcon} ${v.stat} ${lastMaintDate}
${total > 0 ? `
📊 نسبة الجاهزية: ${eff}%
✅ ${good} صالح | 🔧 ${maint} صيانة | ❌ ${bad} معطب
` : ''}
`; }); container.innerHTML = html; } // ============================================================ // دوال الفلترة // ============================================================ function showAllEfficiency() { renderEfficiency(); showAlert('✅ تم عرض جميع الجداول', 'success'); } function filterEfficiencyByUnit() { const select = document.getElementById('effUnitFilter'); if (!select) return; const value = select.value; if (value) { const filtered = allVessels.filter(v => v.reg === value); showAlert(`🔍 عرض إقليم: ${value}`, 'info'); updateEfficiencyStats(filtered); renderGeneralEfficiencyTable(filtered); renderCategoryEfficiencyTable(filtered); renderRegionEfficiencyTables(filtered); } else { showAlert('✅ عرض جميع الأقاليم', 'success'); renderEfficiency(); } } // تصدير الدوال window.initEfficiency = initEfficiency; window.renderEfficiency = renderEfficiency; window.showAllEfficiency = showAllEfficiency; window.filterEfficiencyByUnit = filterEfficiencyByUnit;
