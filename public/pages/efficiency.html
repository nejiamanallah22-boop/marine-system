// public/js/pages/efficiency.js console.log('✅ Efficiency page loaded'); // ============================================================ // تهيئة الصفحة // ============================================================ function initEfficiency() { console.log('📊 Initializing efficiency page'); loadVessels(); } // ============================================================ // الدالة الرئيسية // ============================================================ function renderEfficiency() { console.log('📊 Rendering efficiency, vessels:', allVessels.length); const vessels = allVessels || []; const countEl = document.getElementById('effCount'); if (countEl) countEl.textContent = `📊 ${vessels.length} مركب`; updateEfficiencyStats(vessels); renderGeneralEfficiencyTable(vessels); renderCategoryEfficiencyTable(vessels); renderAllRegionsTables(vessels); } // ============================================================ // الإحصائيات // ============================================================ function updateEfficiencyStats(vessels) { const statsContainer = document.getElementById('statsCards'); if (!statsContainer) return; const total = vessels.length; const good = vessels.filter(v => v.stat === 'صالح').length; const bad = vessels.filter(v => v.stat === 'معطب').length; const maint = vessels.filter(v => v.stat === 'صيانة').length; const eff = total > 0 ? Math.round((good / total) * 100) : 0; statsContainer.innerHTML = `

${good}

✅ صالح

${bad}

❌ معطب

${maint}

🔧 صيانة

${eff}%

📊 الجاهزية

`; } // ============================================================ // 1. الجدول العام // ============================================================ function renderGeneralEfficiencyTable(vessels) { const container = document.getElementById('generalEffTableContainer'); if (!container) return; if (!vessels || vessels.length === 0) { container.innerHTML = `
🚫 لا توجد مراكب مسجلة
`; return; } const sorted = [...vessels].sort((a, b) => { const order = { 'صالح': 0, 'صيانة': 1, 'معطب': 2 }; return (order[a.stat] || 3) - (order[b.stat] || 3); }); let html = `
الجدول العام للمراكب (${vessels.length} مركب)
`; sorted.forEach((v, index) => { const statusColor = v.stat === 'صالح' ? '#28a745' : v.stat === 'معطب' ? '#dc3545' : '#ffc107'; const statusIcon = v.stat === 'صالح' ? '✅' : v.stat === 'معطب' ? '❌' : '🔧'; const lastMaintenance = allMaintenance .filter(r => r.vesselId === v.id) .sort((a, b) => new Date(b.date) - new Date(a.date))[0]; const lastMaintDate = lastMaintenance ? new Date(lastMaintenance.date).toLocaleDateString() : '-'; html += ` `; }); html += `
# المركب الرقم الفئة الإقليم المنطقة الحالة آخر صيانة الإجراءات
${index + 1} ${v.name || '-'} ${v.num || '-'} ${v.cat || '-'} ${v.reg || '-'} ${v.zone || '-'} ${statusIcon} ${v.stat} ${lastMaintDate}
`; container.innerHTML = html; } // ============================================================ // 2. النجاعة العامة حسب الفئات // ============================================================ function renderCategoryEfficiencyTable(vessels) { const container = document.getElementById('categoryEffContainer'); if (!container) return; const categories = ['البروق', 'صقور', 'خوافر', 'طوافات', 'زوارق مزدوجة']; let totalAll = 0, goodAll = 0, badAll = 0, maintAll = 0; let html = `
النجاعة العامة حسب الفئات
`; categories.forEach(cat => { const catVessels = vessels.filter(v => v.cat === cat); const total = catVessels.length; const good = catVessels.filter(v => v.stat === 'صالح').length; const bad = catVessels.filter(v => v.stat === 'معطب').length; const maint = catVessels.filter(v => v.stat === 'صيانة').length; const eff = total > 0 ? Math.round((good / total) * 100) : 0; totalAll += total; goodAll += good; badAll += bad; maintAll += maint; const color = eff >= 80 ? '#28a745' : eff >= 50 ? '#ffc107' : '#dc3545'; html += ` `; }); const totalEff = totalAll > 0 ? Math.round((goodAll / totalAll) * 100) : 0; const totalColor = totalEff >= 80 ? '#28a745' : totalEff >= 50 ? '#ffc107' : '#dc3545'; html += `
الفئة ✅ الصالحة ❌ المعطبة 🔧 الصيانة 📊 الإجمالي 📈 النسبة
${cat} ${good} ${bad} ${maint} ${total} ${eff}%
📊 المجموع الكلي ${goodAll} ${badAll} ${maintAll} ${totalAll} ${totalEff}%
📈 نسبة الجاهزية العامة: ${totalEff}% ${totalEff >= 80 ? '✅ ممتاز' : totalEff >= 50 ? '⚠️ متوسط' : '❌ منخفض'}
`; container.innerHTML = html; } // ============================================================ // 3. جداول الأقاليم (كل إقليم في جدول منفصل) // ============================================================ function renderAllRegionsTables(vessels) { // الشمال renderRegionTable('regionNorthContainer', vessels, 'الشمال', '🗺️ الحرس البحري بالشمال'); // الساحل renderRegionTable('regionEastContainer', vessels, 'الساحل', '🗺️ الحرس البحري بالساحل'); // الوسط renderRegionTable('regionCenterContainer', vessels, 'الوسط', '🗺️ الحرس البحري بالوسط'); // الجنوب renderRegionTable('regionSouthContainer', vessels, 'الجنوب', '🗺️ الحرس البحري بالجنوب'); } function renderRegionTable(containerId, vessels, regionKey, regionName) { const container = document.getElementById(containerId); if (!container) return; const regionVessels = vessels.filter(v => v.reg === regionKey); const total = regionVessels.length; const good = regionVessels.filter(v => v.stat === 'صالح').length; const bad = regionVessels.filter(v => v.stat === 'معطب').length; const maint = regionVessels.filter(v => v.stat === 'صيانة').length; const eff = total > 0 ? Math.round((good / total) * 100) : 0; const color = eff >= 80 ? '#28a745' : eff >= 50 ? '#ffc107' : '#dc3545'; if (total === 0) { container.innerHTML = `
🚫 لا توجد مراكب في هذا الإقليم
`; return; } // ترتيب حسب الحالة const sorted = [...regionVessels].sort((a, b) => { const order = { 'صالح': 0, 'صيانة': 1, 'معطب': 2 }; return (order[a.stat] || 3) - (order[b.stat] || 3); }); let html = `
${regionName} (${total} مركب) | ✅ ${good} | 🔧 ${maint} | ❌ ${bad} ${eff}% جاهزية
`; sorted.forEach((v, index) => { const statusColor = v.stat === 'صالح' ? '#28a745' : v.stat === 'معطب' ? '#dc3545' : '#ffc107'; const statusIcon = v.stat === 'صالح' ? '✅' : v.stat === 'معطب' ? '❌' : '🔧'; const lastMaintenance = allMaintenance .filter(r => r.vesselId === v.id) .sort((a, b) => new Date(b.date) - new Date(a.date))[0]; const lastMaintDate = lastMaintenance ? new Date(lastMaintenance.date).toLocaleDateString() : '-'; html += ` `; }); html += `
# المركب الرقم الفئة المنطقة الحالة آخر صيانة الإجراءات
${index + 1} ${v.name || '-'} ${v.num || '-'} ${v.cat || '-'} ${v.zone || '-'} ${statusIcon} ${v.stat} ${lastMaintDate}
📊 نسبة الجاهزية: ${eff}%
`; container.innerHTML = html; } // ============================================================ // دوال الفلترة // ============================================================ function showAllEfficiency() { renderEfficiency(); showAlert('✅ تم عرض جميع الجداول', 'success'); } function filterEfficiencyByUnit() { const select = document.getElementById('effUnitFilter'); if (!select) return; const value = select.value; const vessels = allVessels || []; if (value) { const filtered = vessels.filter(v => v.reg === value); showAlert(`🔍 عرض إقليم: ${value}`, 'info'); // تحديث الإحصائيات updateEfficiencyStats(filtered); // تحديث الجدول العام (يعرض المراكب المفلترة) renderGeneralEfficiencyTable(filtered); // تحديث جدول الفئات renderCategoryEfficiencyTable(filtered); // تحديث جداول الأقاليم (تظهر فقط الإقليم المختار) renderAllRegionsTables(filtered); } else { showAlert('✅ عرض جميع الأقاليم', 'success'); renderEfficiency(); } } // ============================================================ // دوال مساعدة // ============================================================ function viewVesselMaintenance(vesselId) { const vessel = allVessels.find(v => v.id == vesselId); if (!vessel) { showAlert('❌ المركب غير موجود', 'danger'); return; } const records = allMaintenance.filter(r => r.vesselId == vesselId); if (records.length === 0) { showAlert(`🚫 لا توجد سجلات صيانة للمركب: ${vessel.name}`, 'info'); return; } const modal = document.createElement('div'); modal.style.cssText = ` position: fixed; top: 0; left: 0; right: 0; bottom: 0; background: rgba(0,0,0,0.7); display: flex; justify-content: center; align-items: center; z-index: 99999; `; let html = `

📋 سجل صيانة ${vessel.name}

`; records.slice().reverse().forEach((r, i) => { html += `
#${i+1} ${new Date(r.date).toLocaleString()}
🔧 ${r.description} ${r.status}
🏭 ${r.unit || 'غير محدد'}
👨‍🔧 ${r.technician || 'غير محدد'}
${r.cost ? `
💰 ${r.cost} د.ت
` : ''} ${r.parts && r.parts.length ? `
🔩 قطع الغيار: ${r.parts.map(p => `${p.name}(${p.quantity})`).join(', ')}
` : ''}
`; }); html += `
`; modal.innerHTML = html; document.body.appendChild(modal); } // ============================================================ // تصدير الدوال // ============================================================ window.initEfficiency = initEfficiency; window.renderEfficiency = renderEfficiency; window.showAllEfficiency = showAllEfficiency; window.filterEfficiencyByUnit = filterEfficiencyByUnit; window.viewVesselMaintenance = viewVesselMaintenance;
